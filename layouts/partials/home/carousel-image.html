{{/* Homepage carousel image pipeline. Expects: Path, Alt, Class. Produces capped WebP variants, intrinsic dimensions and a 24px LQIP. */}} {{- $path := .Path | default "" -}} {{- $alt := .Alt | default "" -}} {{- $class := .Class | default "" -}} {{- $image := resources.Get $path -}} {{- if $image -}} {{- $oneWidth := 600 -}} {{- if lt $image.Width $oneWidth }}{{ $oneWidth = $image.Width }}{{ end -}} {{- $twoWidth := 1200 -}} {{- if lt $image.Width $twoWidth }}{{ $twoWidth = $image.Width }}{{ end -}} {{- $one := $image.Resize (printf "%dx webp" $oneWidth) -}} {{- $two := $image.Resize (printf "%dx webp" $twoWidth) -}} {{- $tinyWidth := 24 -}} {{- if lt $image.Width $tinyWidth }}{{ $tinyWidth = $image.Width }}{{ end -}} {{- $tiny := $image.Resize (printf "%dx webp" $tinyWidth) -}} {{- $lqip := printf "data:image/webp;base64,%s" (base64Encode $tiny.Content) | safeURL -}} {{- $srcset := printf "%s %dw" $one.RelPermalink $one.Width -}} {{- if ne $two.RelPermalink $one.RelPermalink -}} {{- $srcset = printf "%s, %s %dw" $srcset $two.RelPermalink $two.Width -}} {{- end -}} {{- $sizes := "(max-width: 720px) 78vw, (max-width: 980px) 12rem, 18rem" -}} {{/* 高清地址暂不交给预加载扫描器；首屏绘制后由轮播脚本下载并解码。 */}} {{- else -}} {{- errorf "Homepage cover image not found: %s" $path -}} ✦ {{- end -}}
