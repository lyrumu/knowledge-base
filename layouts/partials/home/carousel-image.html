{{/* Homepage carousel image pipeline. Expects: Path, Alt, Class, Eager. Produces capped WebP variants, intrinsic dimensions and a 24px LQIP. */}} {{- $path := .Path | default "" -}} {{- $alt := .Alt | default "" -}} {{- $class := .Class | default "" -}} {{- $eager := .Eager | default false -}} {{- $image := resources.Get $path -}} {{- if $image -}} {{- $oneWidth := 600 -}} {{- if lt $image.Width $oneWidth }}{{ $oneWidth = $image.Width }}{{ end -}} {{- $twoWidth := 1200 -}} {{- if lt $image.Width $twoWidth }}{{ $twoWidth = $image.Width }}{{ end -}} {{- $one := $image.Resize (printf "%dx webp" $oneWidth) -}} {{- $two := $image.Resize (printf "%dx webp" $twoWidth) -}} {{- $tinyWidth := 24 -}} {{- if lt $image.Width $tinyWidth }}{{ $tinyWidth = $image.Width }}{{ end -}} {{- $tiny := $image.Resize (printf "%dx webp" $tinyWidth) -}} {{- $lqip := printf "data:image/webp;base64,%s" (base64Encode $tiny.Content) | safeURL -}} {{- else -}} {{- errorf "Homepage cover image not found: %s" $path -}} ✦ {{- end -}}
